def second_largest(lst):
    first = second = float('-inf') # -infinity

    for num in lst:
        if num > first:
            second = first
            first = num
        elif num > second and num!= first:
            second = num

    if second == float('-inf'):
        return "No second largest element"
    return second

L = list(map(int, input("Enter list input: ").split()))
print("Second Largest:", second_largest(L))
